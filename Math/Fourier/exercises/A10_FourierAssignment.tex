\item
Evaluate the Laplace transform of the given functions.
\begin{enumerate}
\item
$f(t)=u(t-1)-t+1$
\item
$f(t)=e^{2t}u(t-2)$
\end{enumerate}

\textbf{Solution:} \\
% ====== TODO =======
% Please write your solution here.

We use the standard transforms
\[
\mathcal{L}\{u(t-a)\}=\frac{e^{-as}}{s},\qquad
\mathcal{L}\{1\}=\frac1s,\qquad
\mathcal{L}\{t\}=\frac{1}{s^{2}},
\]
and the second shifting theorem (shift on the $t$-axis)
\[
\mathcal{L}\{u(t-a)\,g(t-a)\}=e^{-as}\,G(s),\qquad G(s)=\mathcal{L}\{g(t)\}.
\]

\textbf{(a)} Write $-t+1=-(t-1)$, so $f(t)=u(t-1)-(t-1)$. Term by term,
\begin{align*}
\mathcal{L}\{u(t-1)\}&=\frac{e^{-s}}{s},\\
\mathcal{L}\{t-1\}&=\frac{1}{s^{2}}-\frac1s .
\end{align*}
Therefore
\[
\boxed{\;F(s)=\frac{e^{-s}}{s}+\frac1s-\frac{1}{s^{2}}\;}
\]

\textbf{(b)} To match the form $u(t-2)\,g(t-2)$, rewrite the exponential about $t=2$:
\[
e^{2t}=e^{2(t-2)+4}=e^{4}\,e^{2(t-2)},
\qquad\text{so}\qquad
f(t)=e^{4}\,e^{2(t-2)}\,u(t-2).
\]
Here $a=2$ and $g(t)=e^{2t}$ with $G(s)=\dfrac{1}{s-2}$. By the second shifting theorem,
\[
\boxed{\;F(s)=e^{4}\cdot e^{-2s}\cdot\frac{1}{s-2}=\frac{e^{\,4-2s}}{s-2}\;}\qquad(s>2).
\]
(Check by definition: $\displaystyle\int_{2}^{\infty}e^{2t}e^{-st}\,dt=\frac{e^{2(2-s)}}{s-2}=\frac{e^{4-2s}}{s-2}$.)

\item
Evaluate the inverse Laplace transform of the given function.\\
$\mathcal{L}^{-1}[\frac{e^{-s}}{s^2}]$

\textbf{Solution:} \\
% ====== TODO =======
% Please write your solution here.

Since $\mathcal{L}^{-1}\!\left\{\dfrac{1}{s^{2}}\right\}=t$, apply the inverse second
shifting theorem $\mathcal{L}^{-1}\{e^{-as}F(s)\}=u(t-a)\,f(t-a)$ with $a=1$ and $f(t)=t$:
\[
\boxed{\;\mathcal{L}^{-1}\!\left\{\frac{e^{-s}}{s^{2}}\right\}=(t-1)\,u(t-1)
=\begin{cases}0,&t<1,\\[2pt] t-1,&t\ge 1.\end{cases}}
\]
